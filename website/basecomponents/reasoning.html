Reasoning

The reasoning contract defines the loop that drives one Agent turn.

Base class and contract methods#

BaseReasoning is the turn-loop contract. The Agent binds it to the client, dispatcher, and middleware for each run. The subclass implements the loop that requests model output, handles tool calls, and yields runtime events.

BaseReasoning · complete class API
class BaseReasoning(ComponentBase[ReasoningConfig], ABC):
    """Abstract reasoning cycle."""
    component_type = "reasoning"

    component_schema: t.ClassVar[type[BaseModel]] = ReasoningConfig

    LOOP_STATE_CLS: t.ClassVar[type[BaseLoopState]] = BaseLoopState

    def __init__(
        self, max_connection_retries: int = 3, enable_human_input: bool = True
    ) -> None:
        """Initialize config-only state. The base class provides this behavior;
            override it only to customize it."""
        ...

    def _set_loop_state(self, state: BaseLoopState) -> None:
        """Set the loop state shared with runtime callbacks. The base class provides
            this behavior; override it only to customize it."""
        ...

    def bind(
        self,
        name: str,
        client: CoreChatCompletionClient,
        middleware_chain: MiddlewareChain | None = None,
        compaction: CoreCompaction | None = None,
        max_context_tokens: int = 0,
        *,
        dispatcher: ToolDispatcher | None = None,
        tool_context: ToolContext | None = None,
        completion_bus: EventBus | None = None,
        memory: CoreMemoryRegistry | None = None,
    ) -> t.Self:
        """A copy of this loop wired to one run's dependencies (client, tool
            context, memory...). Called by the agent inside ``run()``. The base
            class provides this behavior; override it only to customize it."""
        ...

    def _refresh_session_state(self, ctx: RunContext, prompts: PromptCtx) -> None:
        """Keep the prompt's session state (compaction summary, plan) in sync with
            ``ctx``. Re-renders the stack only when one of them changed. The base
            class provides this behavior; override it only to customize it."""
        ...

    @property
    def _prompt_counter(self) -> TokenCounter:
        """Counter with the client's tokenizer, built once per loop. The base class
            provides this behavior; override it only to customize it."""
        ...

    async def _compact_if_needed(
        self,
        ctx: RunContext,
        prompts: PromptCtx,
    ) -> t.AsyncGenerator[CoreEvent, None]:
        """Run the strategy before a model call and apply its result to ``ctx``. The
            base class provides this behavior; override it only to customize it."""
        ...

    @property
    def name(self) -> str:
        """Return the name of the agent bound to this reasoning loop. The base class
            provides this behavior; override it only to customize it."""
        ...

    @property
    def client(self) -> CoreChatCompletionClient:
        """Return the chat completion client bound to this loop. The base class
            provides this behavior; override it only to customize it."""
        ...

    @property
    def dispatcher(self) -> ToolDispatcher:
        """Return the tool dispatcher bound to this loop. The base class provides
            this behavior; override it only to customize it."""
        ...

    @property
    def middleware_chain(self) -> MiddlewareChain:
        """Return the middleware chain bound to this loop. The base class provides
            this behavior; override it only to customize it."""
        ...

    def _middleware_context(self, ctx: RunContext) -> MiddlewareContext:
        """Build middleware context for the current run. The base class provides
            this behavior; override it only to customize it."""
        ...

    @property
    def completion_bus(self) -> EventBus:
        """Return the event bus used for completion events. The base class provides
            this behavior; override it only to customize it."""
        ...

    @property
    def _tools(self) -> list[CoreTool]:
        """Tools currently available to the agent. The base class provides this
            behavior; override it only to customize it."""
        ...

    @property
    def tool_catalog(self) -> dict[str, CoreTool]:
        """Return the tools indexed by their model-visible names. The base class

        ...

    async def _execute_tools(
        self,

        ...

    @staticmethod
    def _conversation_messages(ctx: RunContext) -> list[CoreMessage]:

    @staticmethod
    def _without_assistant_thinking(

        ...

    def _model_input_messages(self, ctx: RunContext) -> list[CoreMessage]:

        """Return model input messages with per-message token counts attached. The

        ...

    async def _call_llm(
        self,
        ctx: RunContext,
        prompts: PromptCtx,
        loop_state: BaseLoopState,
        cancellation_token: CancellationToken | None = None,
        output_format: type[BaseModel] | None = None,

        tools_override: t.Sequence[CoreTool] | None = None,

            provides this behavior; override it only to customize it."""
        ...

    async def _call_llm_stream(
        self,
        ctx: RunContext,
        prompts: PromptCtx,
        loop_state: BaseLoopState,
        cancellation_token: CancellationToken | None = None,
        output_format: type[BaseModel] | None = None,

        tools_override: t.Sequence[CoreTool] | None = None,
        **kwargs: t.Any,
    ) -> t.AsyncGenerator[CoreEvent, None]:

            this behavior; override it only to customize it."""
        ...

    @staticmethod
    def _merge_tool_call_chunk(
        accumulated: dict[str, dict[str, t.Any]],
        fragment: dict[str, t.Any],
    ) -> None:
        """Merge an incoming tool-call fragment into the accumulator. The base class
            provides this behavior; override it only to customize it."""
        ...

    @staticmethod
    def _build_tool_calls_from_chunks(
        accumulated: dict[str, dict[str, t.Any]],
        _log: ScopedLogger,
    ) -> list["ToolCall"]:
        """Convert accumulated chunk fragments into ``ToolCall`` instances. The base
            class provides this behavior; override it only to customize it."""
        ...

    @abstractmethod
    async def execute_reasoning_loop(
        self,
        ctx: RunContext,
        prompts: PromptCtx,
        loop_state: BaseLoopState,
        stream_tokens: bool = False,
        cancellation_token: CancellationToken | None = None,
        output_format: t.Type[BaseModel] | None = None,
        eval_criteria: list[str] | None = None,
        **kwargs: t.Any,
    ) -> t.AsyncGenerator[CoreEvent, None]:
        """Execute the reasoning loop for one agent turn. A concrete subclass must
            implement this method."""
        ...
MethodWhat it receives or returns
ctxCurrent conversation messages, tool state, user/session IDs, and runtime state.
promptsRendered prompt layers that will be supplied to the model.
loop_stateMutable per-run state for loop progress and counters.
yielded eventsCoreEvent objects that describe model, reasoning, and tool activity.
