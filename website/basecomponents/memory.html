Memory

The memory contract provides user-scoped operations while leaving persistence to a subclass.

Base class and contract methods#

CoreMemoryRegistry is the shared memory contract, not a storage backend. It binds operations to a user and session, provides the agent-facing memory tools, and handles the common memory behavior. A subclass implements connection and persistence/search hooks.

CoreMemoryRegistry · complete class API
class CoreMemoryRegistry(CoreAgentCapabilities[BaseModel], ABC):
    """Shared memory operations; concrete backends own persistence and search."""
    def __init__(
        self,
        user_id: str | None = None,
        session_id: str | None = None,
        tool_mode: MemoryToolMode = MemoryToolMode.FULL,
        *,
        context_days: int | None = 30,
        embedding: CoreEmbedding | None = DEFAULT_EMBEDDING,
    ) -> None:
        """Initialize memory options before binding the registry to a run. The base
            class provides this behavior; override it only to customize it."""
        ...

    def bind(self, user_id: str, session_id: str) -> t.Self:
        """A copy scoped to ``user_id``/``session_id`` that shares this registry's
            backend connection (connect the registry first so every copy reuses one
            client). The base class provides this behavior; override it only to
            customize it."""
        ...

    def _set_scope(self, user_id: str | None, session_id: str | None) -> None:
        """Bind memory operations to one user and session. The base class provides
            this behavior; override it only to customize it."""
        ...

    def _validate_scope(self) -> None:
        """Backend hook: reject ids it can't store safely (e.g. as paths). The base
            class provides this behavior; override it only to customize it."""
        ...

    def _require_scope(self) -> None:
        """Require user and session identifiers before accessing memory. The base
            class provides this behavior; override it only to customize it."""
        ...

    @abstractmethod
    async def connect(self) -> None:
        """Open the registry storage connection. A concrete subclass must implement
            this method."""
        ...

    @abstractmethod
    async def disconnect(self) -> None:
        """Close the registry storage connection. A concrete subclass must implement
            this method."""
        ...

    @abstractmethod
    async def _read_session(self) -> list[MemoryRecord]:
        """Read all categories for this user/session, without a date filter. A
            concrete subclass must implement this method."""
        ...

    @abstractmethod
    async def _write_memory(self, record: MemoryRecord) -> bool:
        """Atomically create or replace a category in this user/session. A concrete
            subclass must implement this method."""
        ...

    @abstractmethod
    async def _delete_memory(self, category: str) -> bool:
        """Delete this session's category and any indexed chunks. A concrete
            subclass must implement this method."""
        ...

    @abstractmethod
    async def _search_memory(self, text: str) -> list[MemorySearchResult]:
        """Return relevant chunks from OTHER sessions of this SAME user. A concrete
            subclass must implement this method."""
        ...

    async def get_context(self) -> list[MemoryRecord]:
        """Current-session memories, newest first, within ``context_days``. The base
            class provides this behavior; override it only to customize it."""
        ...

    async def list_category(self) -> list[str]:
        """List ALL stored categories in this session, including older ones. The
            base class provides this behavior; override it only to customize it."""
        ...

    async def create_or_update(self, category: str, memory: str) -> str:
        """Replace the entire category content or create it, independently of
            compaction. The base class provides this behavior; override it only to
            customize it."""
        ...

    async def delete_memory(self, category: str) -> str:
        """Delete a category only in the bound session. The base class provides this
            behavior; override it only to customize it."""
        ...

    semantic_min_score: t.ClassVar[float] = 0.3

    semantic_limit: t.ClassVar[int] = 20

    async def _rank_by_meaning(
        self, text: str, candidates: t.Sequence[MemorySearchResult],
    ) -> list[MemorySearchResult]:
        """The candidates closest in meaning to ``text`` (needs ``embedding``). The
            base class provides this behavior; override it only to customize it."""
        ...

    async def search_memory(self, text: str) -> list[MemorySearchResult]:
        """Search other sessions without adding their memories to this session. The
            base class provides this behavior; override it only to customize it."""
        ...

    def as_tools(self) -> list[CoreTool]:
        """Build the tools enabled by this registry configuration. The base class
            provides this behavior; override it only to customize it."""
        ...

    @property
    def tools(self) -> list[CoreTool]:
        """Return the memory tools exposed to the agent. The base class provides
            this behavior; override it only to customize it."""
        ...

    def _for_run(self, context: ToolContext | None) -> t.Self:
        """The registry scoped to the calling run (tools are built once). Without a
            run (a tool called directly) the registry's own scope applies. The base
            class provides this behavior; override it only to customize it."""
        ...

    @staticmethod
    def _validate_non_empty(field: str, value: str) -> str:
        """Validate that a text field contains non-whitespace content. The base
            class provides this behavior; override it only to customize it."""
        ...
MethodWhat it receives or returns
connect() / disconnect()Open and close backend resources.
_read_session()Reads all memory categories for the bound user and session.
_write_memory()Creates or replaces one category atomically.
_delete_memory()Deletes one category and any search index entries.
_search_memory()Finds relevant records from other sessions belonging to the same user.
